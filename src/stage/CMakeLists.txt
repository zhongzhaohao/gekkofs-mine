set(CMAKE_CXX_STANDARD 17)
add_executable(gkfs_stage)

  PRIVATE stage.cpp
          stage_util.cpp
)

target_link_libraries(gkfs_stage
    PRIVATE gkfs_user_lib
    # external libs
    CLI11::CLI11
)
